Use this OTS exactly as per the requirement. It creates the web app `datacenter-webapp` in `centralus` under the default resource group, uses Python on Linux, creates the App Service Plan `datacenter-learn-python` with Basic B1, disables Application Insights, adds the required tags, and ensures the web app is running. [learn.microsoft](https://learn.microsoft.com/en-us/cli/azure/webapp?view=azure-cli-latest)

## OTS

```bash
#!/usr/bin/env bash
set -euo pipefail

echo "=== STEP 0: Azure login ==="
az account show >/dev/null 2>&1 || az login --use-device-code >/dev/null

WEBAPP_NAME="datacenter-webapp"
PLAN_NAME="datacenter-learn-python"
LOCATION="centralus"
SKU="B1"
TAG_NAME="WebAppLearning"
TAG_ENV="Dev"

echo
echo "=== STEP 1: Discover default resource group dynamically ==="
RG="$(az group list --query "[0].name" -o tsv)"

if [[ -z "${RG}" ]]; then
  echo "ERROR: Unable to determine default resource group."
  exit 1
fi

echo "Resource Group : ${RG}"
echo "Web App        : ${WEBAPP_NAME}"
echo "Plan           : ${PLAN_NAME}"
echo "Location       : ${LOCATION}"
echo "SKU            : ${SKU}"

echo
echo "=== STEP 2: Pick Python runtime dynamically ==="
PY_RUNTIME="$(az webapp list-runtimes --os linux --query "[?starts_with(@, 'PYTHON:')][-1]" -o tsv)"
if [[ -z "${PY_RUNTIME}" ]]; then
  PY_RUNTIME="PYTHON:3.11"
fi
echo "Selected Runtime: ${PY_RUNTIME}"

echo
echo "=== STEP 3: Create App Service Plan ==="
az appservice plan create \
  --resource-group "${RG}" \
  --name "${PLAN_NAME}" \
  --location "${LOCATION}" \
  --is-linux \
  --sku "${SKU}" \
  -o table

echo
echo "=== STEP 4: Create Linux Python Web App ==="
az webapp create \
  --resource-group "${RG}" \
  --plan "${PLAN_NAME}" \
  --name "${WEBAPP_NAME}" \
  --runtime "${PY_RUNTIME}" \
  -o table

echo
echo "=== STEP 5: Disable Application Insights ==="
az webapp config appsettings delete \
  --resource-group "${RG}" \
  --name "${WEBAPP_NAME}" \
  --setting-names APPINSIGHTS_INSTRUMENTATIONKEY APPLICATIONINSIGHTS_CONNECTION_STRING \
  -o table || true

echo
echo "=== STEP 6: Add required tags ==="
WEBAPP_ID="$(az webapp show -g "${RG}" -n "${WEBAPP_NAME}" --query id -o tsv)"

az tag update \
  --resource-id "${WEBAPP_ID}" \
  --operation Merge \
  --tags Name="${TAG_NAME}" Environment="${TAG_ENV}" \
  -o table

echo
echo "=== STEP 7: Ensure app is running ==="
az webapp start -g "${RG}" -n "${WEBAPP_NAME}" >/dev/null 2>&1 || true

for i in {1..18}; do
  STATE="$(az webapp show -g "${RG}" -n "${WEBAPP_NAME}" --query state -o tsv)"
  echo "Current state: ${STATE}"
  [[ "${STATE}" == "Running" ]] && break
  sleep 10
done

FINAL_STATE="$(az webapp show -g "${RG}" -n "${WEBAPP_NAME}" --query state -o tsv)"

if [[ "${FINAL_STATE}" != "Running" ]]; then
  echo "ERROR: Web app did not reach Running state."
  exit 1
fi

echo
echo "=== SUCCESS ==="
echo "Web App ${WEBAPP_NAME} is running."
```
