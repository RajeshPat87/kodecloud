Use this dynamic OTS. It creates the destination container as **private**, copies only `nautilus.txt` from the source container to the destination container, verifies the file exists in **both** containers, and confirms the downloaded contents are identical. [learn.microsoft](https://learn.microsoft.com/en-us/cli/azure/storage/blob?view=azure-cli-latest)

## OTS

```bash
#!/usr/bin/env bash
set -euo pipefail

echo "=== STEP 0: Azure login ==="
az account show >/dev/null 2>&1 || az login --use-device-code >/dev/null

STORAGE_ACCOUNT="nautilusst3194"
SOURCE_CONTAINER="nautilus-source-22617"
DEST_CONTAINER="nautilus-dest-22346"
BLOB_NAME="nautilus.txt"

echo
echo "=== STEP 1: Get storage account key dynamically ==="
ACCOUNT_KEY="$(az storage account keys list \
  --account-name "${STORAGE_ACCOUNT}" \
  --query "[0].value" -o tsv)"

if [[ -z "${ACCOUNT_KEY}" ]]; then
  echo "ERROR: Unable to retrieve storage account key."
  exit 1
fi

echo "Storage Account  : ${STORAGE_ACCOUNT}"
echo "Source Container : ${SOURCE_CONTAINER}"
echo "Dest Container   : ${DEST_CONTAINER}"
echo "Blob Name        : ${BLOB_NAME}"

echo
echo "=== STEP 2: Create new private destination container ==="
az storage container create \
  --name "${DEST_CONTAINER}" \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --public-access off \
  -o table

echo
echo "=== STEP 3: Copy blob from source container to destination container ==="
az storage blob copy start \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --destination-container "${DEST_CONTAINER}" \
  --destination-blob "${BLOB_NAME}" \
  --source-account-name "${STORAGE_ACCOUNT}" \
  --source-account-key "${ACCOUNT_KEY}" \
  --source-container "${SOURCE_CONTAINER}" \
  --source-blob "${BLOB_NAME}" \
  -o table

echo
echo "=== STEP 4: Wait for copy completion ==="
for i in {1..30}; do
  COPY_STATUS="$(az storage blob show \
    --account-name "${STORAGE_ACCOUNT}" \
    --account-key "${ACCOUNT_KEY}" \
    --container-name "${DEST_CONTAINER}" \
    --name "${BLOB_NAME}" \
    --query "properties.copy.status" -o tsv 2>/dev/null || true)"

  echo "Copy status: ${COPY_STATUS}"
  [[ "${COPY_STATUS}" == "success" ]] && break
  sleep 5
done

FINAL_COPY_STATUS="$(az storage blob show \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --container-name "${DEST_CONTAINER}" \
  --name "${BLOB_NAME}" \
  --query "properties.copy.status" -o tsv)"

if [[ "${FINAL_COPY_STATUS}" != "success" ]]; then
  echo "ERROR: Blob copy did not complete successfully."
  exit 1
fi

echo
echo "=== STEP 5: Verify blob exists in both containers ==="
SRC_EXISTS="$(az storage blob exists \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --container-name "${SOURCE_CONTAINER}" \
  --name "${BLOB_NAME}" \
  --query exists -o tsv)"

DEST_EXISTS="$(az storage blob exists \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --container-name "${DEST_CONTAINER}" \
  --name "${BLOB_NAME}" \
  --query exists -o tsv)"

echo "Source Exists : ${SRC_EXISTS}"
echo "Dest Exists   : ${DEST_EXISTS}"

if [[ "${SRC_EXISTS}" != "true" || "${DEST_EXISTS}" != "true" ]]; then
  echo "ERROR: Blob not present in both containers."
  exit 1
fi

echo
echo "=== STEP 6: Download source and destination blobs for content verification ==="
rm -f /tmp/nautilus_source.txt /tmp/nautilus_dest.txt

az storage blob download \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --container-name "${SOURCE_CONTAINER}" \
  --name "${BLOB_NAME}" \
  --file /tmp/nautilus_source.txt \
  --overwrite \
  -o none

az storage blob download \
  --account-name "${STORAGE_ACCOUNT}" \
  --account-key "${ACCOUNT_KEY}" \
  --container-name "${DEST_CONTAINER}" \
  --name "${BLOB_NAME}" \
  --file /tmp/nautilus_dest.txt \
  --overwrite \
  -o none

echo
echo "=== STEP 7: Compare file content ==="
if cmp -s /tmp/nautilus_source.txt /tmp/nautilus_dest.txt; then
  echo "Verification successful: file content is identical."
else
  echo "ERROR: File content differs between source and destination."
  exit 1
fi

echo
echo "=== SUCCESS ==="
echo "Container '${DEST_CONTAINER}' created as private."
echo "Blob '${BLOB_NAME}' copied successfully."
echo "Both containers contain '${BLOB_NAME}' and contents are identical."
```

## Runbook

1. Save the script as `script.sh`.
2. Run `chmod +x script.sh`.
3. Execute `./script.sh`.
4. Confirm the final success message shows the destination container was created, the blob was copied, and content verification passed.[user query]

## Documentation

This script uses Azure CLI container creation with `--public-access off`, which creates a private container. It uses Azure CLI blob copy operations and then verifies transfer integrity by checking blob existence in both containers and comparing the downloaded file contents byte-for-byte.
