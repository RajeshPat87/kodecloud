## OTS — run on jump host as thor

```bash
#!/usr/bin/env bash
# jump host — complete httpd role: set host, add jinja2 template + copy task
set -e

cd /home/thor/ansible

[ -f ansible.cfg ] || cat > ansible.cfg <<'EOF'
[defaults]
host_key_checking = False
inventory = inventory
EOF

# a. playbook: run httpd role on App Server 1 (keep the existing role reference)
cat > playbook.yml <<'EOF'
---
- hosts: stapp01
  become: yes
  become_user: root
  roles:
    - role/httpd
EOF

# b. jinja2 template using inventory_hostname (not hardcoded)
cat > role/httpd/templates/index.html.j2 <<'EOF'
This file was created using Ansible on {{ inventory_hostname }}
EOF

# c+d. append template-copy task (mode 0655, owner/group tony) to existing tasks
cat >> role/httpd/tasks/main.yml <<'EOF'

- name: Deploy index.html from jinja2 template
  template:
    src: index.html.j2
    dest: /var/www/html/index.html
    owner: tony
    group: tony
    mode: '0655'
EOF

echo "== playbook.yml =="; cat playbook.yml
echo "== template =="; cat role/httpd/templates/index.html.j2
echo "== tasks/main.yml =="; cat role/httpd/tasks/main.yml
```
